  <script type="module">

  

    const SUPABASE_URL = "https://jjiywxidulgiythucivk.supabase.co";

    const SUPABASE_ANON_KEY = "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6ImpqaXl3eGlkdWxnaXl0aHVjaXZrIiwicm9sZSI6ImFub24iLCJpYXQiOjE3NjM2MzkwNjAsImV4cCI6MjA3OTIxNTA2MH0.N-ion8iRnXyyqPJTJ5Y2hSC1NF8IEM0HaAml6W3O-nk";

 

    import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

 

    const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

 

    // --- Références DOM ---

    const authSection = document.getElementById("auth-section");

    const appSection = document.getElementById("app-section");

    const userInfo = document.getElementById("user-info");

 

    const authEmail = document.getElementById("auth-email");

    const authPassword = document.getElementById("auth-password");

    const authMsg = document.getElementById("auth-msg");

 

    const isbnInput = document.getElementById("isbn");

    const titleInput = document.getElementById("title");

    const authorInput = document.getElementById("author");

    const publisherInput = document.getElementById("publisher");

    const yearPublishedInput = document.getElementById("year-published");

    const yearReadInput = document.getElementById("year-read");

    const ratingInput = document.getElementById("rating");

    const statusInput = document.getElementById("status");

    const commentInput = document.getElementById("comment");

    const bookMsg = document.getElementById("book-msg");

    const saveMsg = document.getElementById("save-msg");

    const booksTableBody = document.querySelector("#books-table tbody");

    const familyTableBody = document.querySelector("#family-table tbody");

 

    let currentUser = null;

 

    // --- Affichage auth / appli ---

    async function checkSession() {

      const { data } = await supabase.auth.getSession();

      if (data.session && data.session.user) {

        currentUser = data.session.user;

        showApp();

      } else {

        showAuth();

      }

    }

 

    function showAuth() {

      authSection.style.display = "block";

      appSection.style.display = "none";

      userInfo.innerHTML = "";

    }

 

    function showApp() {

      authSection.style.display = "none";

      appSection.style.display = "block";

      userInfo.innerHTML = `<span class="badge-user">${currentUser.email}</span>`;

      loadBooks();

    }

 

    // --- Auth ---

    document.getElementById("btn-login").addEventListener("click", async () => {

      authMsg.textContent = "";

      authMsg.className = "msg";

      const email = authEmail.value.trim();

      const password = authPassword.value.trim();

      if (!email || !password) {

        authMsg.textContent = "Email et mot de passe obligatoires.";

        authMsg.className = "msg err";

        return;

      }

      const { data, error } = await supabase.auth.signInWithPassword({ email, password });

      if (error) {

        authMsg.textContent = "Erreur de connexion : " + error.message;

        authMsg.className = "msg err";

      } else {

        currentUser = data.user;

        authMsg.textContent = "";

        showApp();

      }

    });

 

    document.getElementById("btn-signup").addEventListener("click", async () => {

      authMsg.textContent = "";

      authMsg.className = "msg";

      const email = authEmail.value.trim();

      const password = authPassword.value.trim();

      if (!email || !password) {

        authMsg.textContent = "Email et mot de passe obligatoires.";

        authMsg.className = "msg err";

        return;

      }

      const { data, error } = await supabase.auth.signUp({ email, password });

      if (error) {

        authMsg.textContent = "Erreur d'inscription : " + error.message;

        authMsg.className = "msg err";

      } else {

        authMsg.textContent = "Compte créé, vérifie ton e-mail si nécessaire, puis connecte-toi.";

        authMsg.className = "msg ok";

      }

    });

 

    document.getElementById("btn-logout").addEventListener("click", async () => {

      await supabase.auth.signOut();

      currentUser = null;

      showAuth();

    });

 

    // --- Récupération infos livre depuis OpenLibrary ---

    document.getElementById("btn-fetch").addEventListener("click", async () => {

      const isbn = isbnInput.value.trim();

      bookMsg.textContent = "";

      bookMsg.className = "msg";

      if (!isbn) {

        bookMsg.textContent = "ISBN vide.";

        bookMsg.className = "msg err";

        return;

      }

      try {

        const res = await fetch(`https://openlibrary.org/isbn/${isbn}.json`);

        if (!res.ok) {

          bookMsg.textContent = "Livre non trouvé dans OpenLibrary.";

          bookMsg.className = "msg err";

          return;

        }

        const data = await res.json();

        titleInput.value = data.title || "";

 

        // Année de sortie

        let yearPublished = "";

        if (data.publish_date) {

          const parts = data.publish_date.split(" ");

          for (const part of parts) {

            if (/^\d{4}$/.test(part)) {

              yearPublished = part;

              break;

            }

          }

        }

        yearPublishedInput.value = yearPublished;

 

        // Éditeur

        if (Array.isArray(data.publishers) && data.publishers.length > 0) {

          if (typeof data.publishers[0] === "string") {

            publisherInput.value = data.publishers[0];

          } else if (typeof data.publishers[0] === "object") {

            publisherInput.value = data.publishers[0].name || "";

          }

        }

 

        // Auteur (appel supplémentaire)

        authorInput.value = "";

        if (Array.isArray(data.authors) && data.authors.length > 0 && data.authors[0].key) {

          try {

            const aRes = await fetch(`https://openlibrary.org${data.authors[0].key}.json`);

            if (aRes.ok) {

              const aData = await aRes.json();

              authorInput.value = aData.name || "";

            }

          } catch (e) {

            // on ignore

          }

        }

 

        bookMsg.textContent = "Infos récupérées, tu peux corriger si besoin.";

        bookMsg.className = "msg ok";

      } catch (e) {

        bookMsg.textContent = "Erreur lors de l'appel OpenLibrary.";

        bookMsg.className = "msg err";

      }

    });

 

    // --- Enregistrer un livre ---

    document.getElementById("btn-save").addEventListener("click", async () => {

      saveMsg.textContent = "";

      saveMsg.className = "msg";

 

      if (!currentUser) {

        saveMsg.textContent = "Tu dois être connecté.";

        saveMsg.className = "msg err";

        return;

      }

 

      const isbn = isbnInput.value.trim();

      const title = titleInput.value.trim() || "Titre inconnu";

      const author = authorInput.value.trim() || "Auteur inconnu";

      const publisher = publisherInput.value.trim() || "Éditeur inconnu";

      const year_published = yearPublishedInput.value.trim() || "Année inconnue";

      const year_read = yearReadInput.value.trim() || "Inconnue";

      const ratingRaw = ratingInput.value.trim();

      const status = statusInput.value || "lu";

      const comment = commentInput.value.trim();

 

      if (!isbn) {

        saveMsg.textContent = "ISBN obligatoire.";

        saveMsg.className = "msg err";

        return;

      }

 

      let rating = null;

      if (ratingRaw) {

        const parsed = parseFloat(ratingRaw.replace(",", "."));

        if (!isNaN(parsed)) rating = parsed;

      }

 

      // Vérifie si un livre avec cet ISBN existe déjà pour cet utilisateur

      const { data: existing, error: selError } = await supabase

        .from("books")

        .select("*")

        .eq("user_id", currentUser.id)

        .eq("isbn", isbn)

        .maybeSingle();

 

      if (selError && selError.code !== "PGRST116") {

        saveMsg.textContent = "Erreur de lecture : " + selError.message;

        saveMsg.className = "msg err";

        return;

      }

 

      if (existing) {

        // update

        const { error: updError } = await supabase

          .from("books")

          .update({

            title,

            author,

            publisher,

            year_published,

            year_read,

            rating,

            comment,

            status

          })

          .eq("id", existing.id);

 

        if (updError) {

          saveMsg.textContent = "Erreur de mise à jour : " + updError.message;

          saveMsg.className = "msg err";

          return;

        }

        saveMsg.textContent = "Livre mis à jour.";

        saveMsg.className = "msg ok";

      } else {

        // insert

        const { error: insError } = await supabase

          .from("books")

          .insert({

            user_id: currentUser.id,

            isbn,

            title,

            author,

            publisher,

            year_published,

            year_read,

            rating,

            comment,

            status

          });

 

        if (insError) {

          saveMsg.textContent = "Erreur d'enregistrement : " + insError.message;

          saveMsg.className = "msg err";

          return;

        }

        saveMsg.textContent = "Livre enregistré.";

        saveMsg.className = "msg ok";

      }

 

      await loadBooks();

    });

 

    // --- Charger les livres de l'utilisateur ---

    async function loadBooks() {

      booksTableBody.innerHTML = "";

      if (!currentUser) return;

      const { data, error } = await supabase

        .from("books")

        .select("*")

        .eq("user_id", currentUser.id)

        .order("created_at", { ascending: false });

 

      if (error) {

        console.error(error);

        return;

      }

      for (const book of data) {

        const tr = document.createElement("tr");

        tr.innerHTML = `

          <td>${book.title || ""}</td>

          <td>${book.author || ""}</td>

          <td>

            <div>ISBN : ${book.isbn || ""}</div>

            <div>Éditeur : ${book.publisher || ""}</div>

            <div>Sortie : ${book.year_published || ""}</div>

          </td>

          <td>

            <div>${book.status || ""}</div>

            <div>Année : ${book.year_read || ""}</div>

          </td>

          <td>${book.rating ?? ""}</td>

          <td>${(book.comment || "").replace(/\n/g, "<br>")}</td>

        `;

        booksTableBody.appendChild(tr);

      }

    }

 

    document.getElementById("btn-refresh").addEventListener("click", loadBooks);

 

    // --- Export CSV (livres de l'utilisateur) ---

    document.getElementById("btn-export").addEventListener("click", async () => {

      if (!currentUser) return;

 

      const { data, error } = await supabase

        .from("books")

        .select("*")

        .eq("user_id", currentUser.id);

 

      if (error) {

        alert("Erreur lors de l'export : " + error.message);

        return;

      }

 

      if (!data || data.length === 0) {

        alert("Aucun livre à exporter.");

        return;

      }

 

      const headers = [

        "isbn","title","author","publisher",

        "year_published","year_read","status","rating","comment"

      ];

 

      const rows = [headers.join(";")];

 

      for (const book of data) {

        const row = [

          book.isbn || "",

          book.title || "",

          book.author || "",

          book.publisher || "",

          book.year_published || "",

          book.year_read || "",

          book.status || "",

          book.rating ?? "",

          (book.comment || "").replace(/\r?\n/g, " ")

        ].map(v => `"${String(v).replace(/"/g, '""')}"`).join(";");

        rows.push(row);

      }

 

      const csvContent = rows.join("\n");

      const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });

      const url = URL.createObjectURL(blob);

 

      const a = document.createElement("a");

      a.href = url;

      a.download = "mes_livres.csv";

      document.body.appendChild(a);

      a.click();

      document.body.removeChild(a);

      URL.revokeObjectURL(url);

    });

 

    // --- Top livres de la famille ---

    async function loadFamilyTopBooks() {

      familyTableBody.innerHTML = "";

 

      // RLS côté Supabase doit autoriser SELECT pour tous les users authentifiés

      const { data, error } = await supabase

        .from("books")

        .select("title, author, rating")

        .not("rating", "is", null);

 

      if (error) {

        console.error(error);

        return;

      }

 

      const map = new Map();

      for (const b of data) {

        const key = `${b.title || ""}|||${b.author || ""}`;

        if (!map.has(key)) {

          map.set(key, { title: b.title || "", author: b.author || "", sum: 0, count: 0 });

        }

        const item = map.get(key);

        item.sum += Number(b.rating) || 0;

        item.count += 1;

      }

 

      const list = [];

      for (const item of map.values()) {

        list.push({

          title: item.title,

          author: item.author,

          count: item.count,

          avg: item.sum / item.count

        });

      }

 

      list.sort((a, b) => {

        if (b.avg !== a.avg) return b.avg - a.avg;

        return b.count - a.count;

      });

 

      const top = list.slice(0, 20);

 

      for (const book of top) {

        const tr = document.createElement("tr");

        tr.innerHTML = `

          <td>${book.title}</td>

          <td>${book.author}</td>

          <td>${book.count}</td>

          <td>${book.avg.toFixed(2)}</td>

        `;

        familyTableBody.appendChild(tr);

      }

    }

 

    document.getElementById("btn-family-top").addEventListener("click", loadFamilyTopBooks);
